cmake_minimum_required(VERSION 3.16)
project(MediaGateway LANGUAGES CXX)

include(CTest)
add_library(mg_rtp_quality src/quality/RtpSequenceStats.cpp)
target_include_directories(mg_rtp_quality PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(mg_rtp_quality PUBLIC cxx_std_17)
if(MSVC)
  target_compile_options(mg_rtp_quality PRIVATE /W4)
else()
  target_compile_options(mg_rtp_quality PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
  add_executable(rtp_sequence_tests tests/rtp_sequence_tests.cpp)
  target_link_libraries(rtp_sequence_tests PRIVATE mg_rtp_quality)
  add_test(NAME rtp_sequence_boundaries COMMAND rtp_sequence_tests)
endif()
